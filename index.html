const { Client, GatewayIntentBits, REST, Routes, SlashCommandBuilder, ActivityType } = require("discord.js"); const client = new Client({ intents: [GatewayIntentBits.Guilds] }); const commands = [ new SlashCommandBuilder() .setName("countdown") .setDescription("See the countdown to a holiday") .addStringOption(option => option.setName("holiday") .setDescription("Choose a holiday") .setRequired(true) .addChoices( { name: "Valentine's Day 💕", value: "valentines" }, { name: "St. Patrick's Day ☘️", value: "stpatricks" }, { name: "Easter 🐣", value: "easter" }, { name: "Halloween 🎃", value: "halloween" }, { name: "Thanksgiving 🦃", value: "thanksgiving" }, { name: "Christmas 🎄", value: "christmas" }, { name: "New Year's Day 🎆", value: "newyear" } ) ), new SlashCommandBuilder() .setName("holidays") .setDescription("See all supported holidays") ].map(command => command.toJSON()); function easterDate(year) { const a = year % 19; const b = Math.floor(year / 100); const c = year % 100; const d = Math.floor(b / 4); const e = b % 4; const f = Math.floor((b + 8) / 25); const g = Math.floor((b - f + 1) / 3); const h = (19 * a + b - d - g + 15) % 30; const i = Math.floor(c / 4); const k = c % 4; const l = (32 + 2 * e + 2 * i - h - k) % 7; const m = Math.floor((a + 11 * h + 22 * l) / 451); const month = Math.floor((h + l - 7 * m + 114) / 31); const day = ((h + l - 7 * m + 114) % 31) + 1; return new Date(year, month - 1, day); } function nthWeekday(year, month, weekday, n) { const first = new Date(year, month, 1); return new Date( year, month, 1 + ((weekday - first.getDay() + 7) % 7) + 7 * (n - 1) ); } function getHolidays(year) { return [ { key: "newyear", name: "New Year's Day", emoji: "🎆", date: new Date(year, 0, 1) }, { key: "valentines", name: "Valentine's Day", emoji: "💕", date: new Date(year, 1, 14) }, { key: "stpatricks", name: "St. Patrick's Day", emoji: "☘️", date: new Date(year, 2, 17) }, { key: "easter", name: "Easter", emoji: "🐣", date: easterDate(year) }, { key: "halloween", name: "Halloween", emoji: "🎃", date: new Date(year, 9, 31) }, { key: "thanksgiving", name: "Thanksgiving", emoji: "🦃", date: nthWeekday(year, 10, 4, 4) }, { key: "christmas", name: "Christmas", emoji: "🎄", date: new Date(year, 11, 25) } ]; } function nextHoliday(key) { const now = new Date(); for (const year of [now.getFullYear(), now.getFullYear() + 1]) { const holiday = getHolidays(year).find(h => h.key === key); if (holiday && holiday.date > now) return holiday; } return null; } function getCountdown(date) { const total = Math.max(0, date.getTime() - Date.now()); const days = Math.floor(total / 86400000); const hours = Math.floor((total % 86400000) / 3600000); const minutes = Math.floor((total % 3600000) / 60000); const seconds = Math.floor((total % 60000) / 1000); return `${days} days, ${hours} hours, ${minutes} minutes, ${seconds} seconds`; } client.once("ready", async () => { console.log(`${client.user.tag} is online!`); const rest = new REST({ version: "10" }) .setToken(process.env.DISCORD_TOKEN); try { await rest.put( Routes.applicationGuildCommands( process.env.CLIENT_ID, process.env.GUILD_ID ), { body: commands } ); console.log("Slash commands registered!"); } catch (error) { console.error("Command registration failed:", error); } function updateStatus() { const now = new Date(); const upcoming = [ ...getHolidays(now.getFullYear()), ...getHolidays(now.getFullYear() + 1) ] .filter(h => h.date > now) .sort((a, b) => a.date - b.date)[0]; if (upcoming) { client.user.setPresence({ activities: [{ name: `${upcoming.emoji} ${upcoming.name} in ${Math.ceil( (upcoming.date - now) / 86400000 )} days`, type: ActivityType.Watching }], status: "online" }); } } updateStatus(); setInterval(updateStatus, 60 * 1000); }); client.on("interactionCreate", async interaction => { if (!interaction.isChatInputCommand()) return; if (interaction.commandName === "countdown") { const key = interaction.options.getString("holiday"); const holiday = nextHoliday(key); if (!holiday) { return interaction.reply({ content: "I couldn't find that holiday!", ephemeral: true }); } return interaction.reply({ embeds: [{ color: 0x9b59b6, title: `${holiday.emoji} Countdown to ${holiday.name}!`, description: `**${getCountdown(holiday.date)}**`, footer: { text: "Countdown to Secret" }, timestamp: new Date().toISOString() }] }); } if (interaction.commandName === "holidays") { const now = new Date(); const holidays = [ ...getHolidays(now.getFullYear()), ...getHolidays(now.getFullYear() + 1) ] .filter(h => h.date > now) .sort((a, b) => a.date - b.date); return interaction.reply({ embeds: [{ color: 0x9b59b6, title: "🎉 Upcoming Holidays", description: holidays.map(h => `${h.emoji} **${h.name}** — ` ).join("\n"), footer: { text: "Countdown to Secret" } }] }); } }); client.login(process.env.DISCORD_TOKEN);
